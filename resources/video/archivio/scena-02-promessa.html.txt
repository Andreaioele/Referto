      // SINCRONIA-INIZIO (generato da voce/sincronizza.py: non modificare a mano)
      // Tempi della scena: [tempo sulla voce di riferimento, tempo sulla traccia].
      const SINCRONIA = [[0.0, 0.0], [0.603, 0.4], [1.188, 1.35], [2.774, 2.42], [4.809, 4.28], [5.38, 5.08], [5.965, 5.73], [7.468, 7.26], [10.513, 11.26]];
      // Sposta ogni tween sulla traccia nuova e ne adatta la durata.
      function riallinea(tl) {
        const T = (t) => {
          if (t <= SINCRONIA[0][0]) return t - SINCRONIA[0][0] + SINCRONIA[0][1];
          for (let i = 1; i < SINCRONIA.length; i++) {
            if (t <= SINCRONIA[i][0]) {
              const [a0, b0] = SINCRONIA[i - 1];
              const [a1, b1] = SINCRONIA[i];
              return b0 + ((t - a0) * (b1 - b0)) / (a1 - a0);
            }
          }
          const [a, b] = SINCRONIA[SINCRONIA.length - 1];
          return b + (t - a);
        };
        for (const figlio of tl.getChildren(false, true, true)) {
          const inizio = figlio.startTime();
          const durata = figlio.totalDuration();
          const nuovoInizio = T(inizio);
          figlio.startTime(nuovoInizio);
          const nuovaDurata = T(inizio + durata) - nuovoInizio;
          if (durata > 0 && nuovaDurata > 0.01) figlio.timeScale((figlio.timeScale() * durata) / nuovaDurata);
        }
      }
      // SINCRONIA-FINE
        // Tempi locali della scena: la voce parte a 0,5 s (audio_meta.json + 0,5).

        // Il cerchio chiaro copre la scena buia (11 volte il suo diametro supera la diagonale).

        // «Questo è Referto»: prima la R, poi il nome che esce da dietro.
        tl.fromTo(
          "#s2-r",
          { opacity: 0, scale: 0.4, rotation: -10 },
          { opacity: 1, scale: 1, rotation: 0, duration: 0.8, ease: "back.out(1.8)" },
          0.6,
        );
        tl.fromTo("#s2-nome", { opacity: 0, x: -120 }, { opacity: 1, x: 0, duration: 0.8, ease: "power3.out" }, 1.15);

        // «Prende il tuo referto, e te lo spiega in italiano semplice»: lo slogan parola per parola.

        document.querySelectorAll("#s2-slogan .s2-parola").forEach((parola, i) => {
          tl.fromTo(
            parola,
            { opacity: 0, y: 28 },
            { opacity: 1, y: 0, duration: 0.5, ease: "power3.out" },

          );
        });
