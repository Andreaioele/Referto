      const SINCRONIA = [[0.0, 0.0], [1.103, 0.4], [2.021, 2.05], [3.444, 3.61], [5.081, 4.86], [7.862, 7.25], [9.338, 10.38], [11.957, 11.45], [12.503, 12.9], [14.763, 15.53]];

          if (t <= SINCRONIA[0][0]) return t - SINCRONIA[0][0] + SINCRONIA[0][1];

            if (t <= SINCRONIA[i][0]) {
              const [a0, b0] = SINCRONIA[i - 1];
              const [a1, b1] = SINCRONIA[i];
              return b0 + ((t - a0) * (b1 - b0)) / (a1 - a0);
            }

          const [a, b] = SINCRONIA[SINCRONIA.length - 1];
          return b + (t - a);

          const durata = figlio.totalDuration();
          const nuovoInizio = T(inizio);
          figlio.startTime(nuovoInizio);
          const nuovaDurata = T(inizio + durata) - nuovoInizio;
          if (durata > 0 && nuovaDurata > 0.01) figlio.timeScale((figlio.timeScale() * durata) / nuovaDurata);
        }
